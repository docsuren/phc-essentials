/* Public Health Calculator: The Essentials
   Queries, feedback and discussion board (the @ icon) and visitor count.
   Created and compiled by Dr Surendran Venkataraman and Dr Angusubalakshmi Rajasivachalam.

   Set up:
   1. Create a new Google Sheet. Open Extensions > Apps Script and paste this whole file.
   2. Save, choose the function "setup" and select Run once (allow the permissions asked).
   3. Deploy > New deployment > type Web app. Execute as: Me. Who has access: Anyone. Deploy.
   4. Copy the web app address (it ends in /exec) into config.json on the website:
      { "api": "https://script.google.com/macros/s/YOUR_ID/exec" }

   To publish a post: type YES in its Approved column.
   To answer: write in the Author reply column, and type YES in Approved.
   To remove: delete the row, or clear Approved.
   The visitor total is kept in cell B1 of the Counter tab. */

const NOTIFY_OWNER = true;      // Sends you an email for each new post. Set to false to stop.
const MAX_POSTS_PER_HOUR = 40;  // Limits automated spam.
const TIME_ZONE = 'Asia/Kolkata';
const APP = 'PHC Essentials';
const TYPES = ['Query', 'Feedback', 'Discussion'];
const HEAD = ['Date', 'ID', 'Reply to', 'Type', 'Name', 'Calculator', 'Message', 'Rating', 'Approved', 'Author reply'];

function setup() {
  const s = sheet_();
  [130, 70, 70, 90, 130, 80, 420, 60, 80, 340].forEach((w, i) => s.setColumnWidth(i + 1, w));
  s.getRange(1, 1, 1, HEAD.length).setFontWeight('bold').setBackground('#dcedf2');
  s.getRange('G:G').setWrap(true); s.getRange('J:J').setWrap(true);
  const rule = SpreadsheetApp.newDataValidation().requireValueInList(['YES', 'NO'], true).setAllowInvalid(true).build();
  s.getRange(2, 9, Math.max(1, s.getMaxRows() - 1), 1).setDataValidation(rule);
  counter_();
  Logger.log(APP + ' board is ready. Email alerts left today: ' + MailApp.getRemainingDailyQuota());
}
function sheet_() {
  const book = SpreadsheetApp.getActive();
  if (!book) throw new Error('Open this script from Extensions > Apps Script inside the Google Sheet.');
  let s = book.getSheetByName('Posts');
  if (!s) { s = book.insertSheet('Posts'); s.appendRow(HEAD); s.setFrozenRows(1); }
  return s;
}
function counter_() {
  const book = SpreadsheetApp.getActive();
  let s = book.getSheetByName('Counter');
  if (!s) { s = book.insertSheet('Counter'); s.getRange('A1').setValue('Visitors'); s.getRange('B1').setValue(0); }
  return s;
}
function count_() { return Number(counter_().getRange('B1').getValue()) || 0; }
function bump_() {
  const lock = LockService.getScriptLock(); lock.waitLock(8000);
  try { const n = count_() + 1; counter_().getRange('B1').setValue(n); return n; }
  finally { lock.releaseLock(); }
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function isYes_(v) { return v === true || String(v).trim().toUpperCase() === 'YES'; }
function clean_(v, max) {
  const t = String(v == null ? '' : v).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').trim().slice(0, max);
  return /^[=+\-@\t\r]/.test(t) ? ' ' + t : t;   // Stops text being treated as a spreadsheet formula.
}
function posts_() {
  const rows = sheet_().getDataRange().getValues().slice(1), out = [];
  rows.forEach(r => {
    if (!isYes_(r[8])) return;
    out.push({
      date: r[0] instanceof Date ? Utilities.formatDate(r[0], TIME_ZONE, 'd MMM yyyy') : String(r[0]),
      id: String(r[1]), parent: String(r[2] || ''), type: String(r[3]), name: String(r[4]).trim(), page: String(r[5] || ''),
      msg: String(r[6]).trim(), rating: Number(r[7]) || 0, reply: String(r[9] || '').trim()
    });
  });
  return out.reverse().slice(0, 300);
}
function doGet(e) {
  try {
    const a = e.parameter.a || '';
    if (a === 'hit') return json_({ count: bump_() });
    if (a === 'count') return json_({ count: count_() });
    if (a === 'posts') return json_({ posts: posts_() });
    return json_({ ok: true, service: APP + ' board' });
  } catch (err) { return json_({ error: 'server' }); }
}
function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.website) return json_({ ok: true });              // Hidden field that only bots fill in.
    const msg = clean_(d.msg, 2000), type = TYPES.indexOf(d.type) >= 0 ? d.type : 'Query';
    if (msg.length < 3) return json_({ error: 'short' });
    const cache = CacheService.getScriptCache(), key = 'posts' + Math.floor(Date.now() / 3600000), used = Number(cache.get(key)) || 0;
    if (used >= MAX_POSTS_PER_HOUR) return json_({ error: 'busy' });
    cache.put(key, String(used + 1), 3700);
    const parent = /^[a-z0-9]{1,16}$/.test(String(d.parent || '')) ? d.parent : '';
    const page = /^[A-Z][0-9]{1,2}$/.test(String(d.page || '')) ? d.page : '';
    const rating = type === 'Feedback' ? Math.min(5, Math.max(0, Math.round(Number(d.rating) || 0))) : '';
    const id = Utilities.getUuid().replace(/-/g, '').slice(0, 12), name = clean_(d.name, 60) || 'Anonymous';
    const lock = LockService.getScriptLock(); lock.waitLock(8000);
    try { sheet_().appendRow([new Date(), id, parent, type, name, page, msg, rating, '', '']); } finally { lock.releaseLock(); }
    if (NOTIFY_OWNER) {
      try { MailApp.sendEmail(Session.getEffectiveUser().getEmail(), APP + ': new ' + type.toLowerCase() + (page ? ' on ' + page : ''), name + ' wrote:\n\n' + msg + '\n\nOpen your Google Sheet and type YES in the Approved column to publish it.'); } catch (err) {}
    }
    return json_({ ok: true });
  } catch (err) { return json_({ error: 'server' }); }
}
